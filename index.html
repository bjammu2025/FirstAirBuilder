window.createNewULD = async function() { const input = document.getElementById("newULDInput"); const uld = input.value .trim() .toUpperCase(); if (!uld) { alert("Enter a ULD / Position ID."); input.focus(); return; } if (!currentUser) { alert("Please enter User Name first."); return; } try { /* Check whether this ULD already exists. ULD = Position ID */ const existingSnapshot = await getDocs( query( collection(db, "ulds"), where("uldId", "==", uld) ) ); let openExists = false; let closedExists = false; existingSnapshot.forEach(item => { const data = item.data(); if (data.status === "OPEN") { openExists = true; } if (data.status === "CLOSED") { closedExists = true; } }); /* Do not allow the same ULD / Position to be opened twice. */ if (openExists) { alert( "ULD / Position ID " + uld + " is already OPEN." ); return; } /* If the ULD was previously closed, don't silently create another one. This prevents duplicate ULD records. */ if (closedExists) { alert( "ULD / Position ID " + uld + " has already been CLOSED.\n\n" + "You cannot create it again." ); return; } /* Create the new ULD. */ const newULDData = { uldId: uld, status: "OPEN", weightLbs: null, weightKg: null, createdBy: currentUser, createdAt: serverTimestamp(), closedAt: null, closedBy: null }; const newULDRef = await addDoc( collection(db, "ulds"), newULDData ); console.log( "ULD created successfully:", newULDRef.id, newULDData ); /* Set current ULD / Position ID. */ currentULD = uld; input.value = ""; /* Go directly to Allocate scan. */ await startAllocateScan(); } catch (error) { console.error( "CREATE ULD ERROR:", error ); /* Show the actual Firebase error. This is important because if Firestore rules are blocking the write, we will immediately know. */ let message = "Could not create ULD."; if (error && error.code) { message += "\n\nFirebase error: " + error.code; } if (error && error.message) { message += "\n" + error.message; } alert(message); } };
